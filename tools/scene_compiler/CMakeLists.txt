cmake_minimum_required(VERSION 3.18.0)
project(scene_compiler)

add_subdirectory("${MAIN_PROJECT_DIRECTORY_PATH}/libraries/easygltf" easygltf)

file(GLOB_RECURSE SCENE_COMPILER_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB_RECURSE SCENE_COMPILER_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.h")

add_executable(${PROJECT_NAME} ${SCENE_COMPILER_SOURCES} ${SCENE_COMPILER_HEADERS})

set_target_properties(${PROJECT_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${TOOLS_BINARIES_PATH}/${PROJECT_NAME}
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${TOOLS_BINARIES_PATH}/${PROJECT_NAME}
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${TOOLS_BINARIES_PATH}/${PROJECT_NAME}
    RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL ${TOOLS_BINARIES_PATH}/${PROJECT_NAME}
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO ${TOOLS_BINARIES_PATH}/${PROJECT_NAME}
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    CXX_EXTENSIONS OFF
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

target_link_libraries(${PROJECT_NAME} 
	PRIVATE easygltf
    PRIVATE utility_lib
)
target_include_directories(${PROJECT_NAME} 
	PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PUBLIC "${MAIN_PROJECT_DIRECTORY_PATH}/libraries/easygltf/include"
)
