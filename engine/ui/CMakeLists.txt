cmake_minimum_required(VERSION 3.18.0)
project(engine_ui)

file(GLOB_RECURSE ENGINE_UI_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB_RECURSE ENGINE_UI_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.h")

set(ENGINE_UI_SOURCES
    ${ENGINE_UI_SOURCES}
    ${ENGINE_UI_HEADERS}
)

add_library(${PROJECT_NAME} STATIC ${ENGINE_UI_SOURCES})

set_target_properties(${PROJECT_NAME} PROPERTIES
    CXX_EXTENSIONS OFF
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

target_include_directories(${PROJECT_NAME} 
	PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/"
    PUBLIC "${MAIN_PROJECT_DIRECTORY_PATH}/engine"
)
