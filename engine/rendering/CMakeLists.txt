cmake_minimum_required(VERSION 3.18.0)
project(engine_rendering)

#add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/DebugDraw")

file(GLOB ENGINE_RENDERING_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB ENGINE_RENDERING_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.h")

file(GLOB ENGINE_RENDERING_GRAPH_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/render_graph/*.cpp")
file(GLOB ENGINE_RENDERING_GRAPH_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/render_graph/*.h")

file(GLOB ENGINE_RENDERING_WORLD_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/render_world/*.cpp")
file(GLOB ENGINE_RENDERING_WORLD_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/render_world/*.h")

set(RENDER_INTERFACE_SOURCES
    ${ENGINE_RENDERING_SOURCES}
    ${ENGINE_RENDERING_HEADERS}

    ${ENGINE_RENDERING_GRAPH_SOURCES}
    ${ENGINE_RENDERING_GRAPH_HEADERS}

    ${ENGINE_RENDERING_WORLD_SOURCES}
    ${ENGINE_RENDERING_WORLD_HEADERS}
)

add_library(${PROJECT_NAME} STATIC ${RENDER_INTERFACE_SOURCES})

set_target_properties(${PROJECT_NAME} PROPERTIES
    CXX_EXTENSIONS OFF
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

target_include_directories(${PROJECT_NAME} 
    PUBLIC "${MAIN_PROJECT_DIRECTORY_PATH}/tools"
    PUBLIC "${MAIN_PROJECT_DIRECTORY_PATH}/engine"
)
