cmake_minimum_required(VERSION 3.18.0)
project(imgui_plugin)

find_package(imgui CONFIG REQUIRED)

file(GLOB_RECURSE IMGUI_PLUGIN_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB_RECURSE IMGUI_PLUGIN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/*.h")

set(PLUGINS_OUTPUT_PATH ${ENGINE_BINARIES_PATH}/embed/rendering/plugins)

set(IMGUI_PLUGIN_SOURCES
    ${IMGUI_PLUGIN_SOURCES}
    ${IMGUI_PLUGIN_HEADERS}
)

add_library(${PROJECT_NAME} SHARED ${RENDER_INTERFACE_SOURCES})
target_compile_definitions(${PROJECT_NAME} PRIVATE -DImGuiPlugin_EXPORTS)

set_target_properties(${PROJECT_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${PLUGINS_OUTPUT_PATH}
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${PLUGINS_OUTPUT_PATH}
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${PLUGINS_OUTPUT_PATH}
    RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL ${PLUGINS_OUTPUT_PATH}
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO ${PLUGINS_OUTPUT_PATH}
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    CXX_EXTENSIONS OFF
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

target_include_directories(${PROJECT_NAME} 
	PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/"
    PUBLIC "${MAIN_PROJECT_DIRECTORY_PATH}/engine"
    PUBLIC "${CMAKE_BINARY_DIR}/exports"
)

target_link_libraries(${PROJECT_NAME}
    PRIVATE imgui::imgui
)

target_compile_definitions(${PROJECT_NAME} 
	PUBLIC "VK_NO_PROTOTYPES"
)
