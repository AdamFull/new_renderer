cmake_minimum_required(VERSION 3.28.0)
project(engine)

if(WIN32)
add_compile_definitions("PLATFORM_WIN32")
endif()

add_compile_definitions(FMT_UNICODE=0)

if(MSVC)
    add_compile_options(/bigobj)
endif()

add_compile_definitions(
    IMGUI_DEFINE_MATH_OPERATORS
    IMGUI_USE_WCHAR32
    NOMINMAX
)

add_compile_definitions($<$<CONFIG:DEBUG>:ENGINE_DEBUG>)

add_subdirectory(engine)

set(APP_SOURCES main.cpp)

if(ANDROID)
    if(CMAKE_VS_NsightTegra_VERSION)
        list(APPEND APP_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/android/AndroidManifest.xml)
    endif()

    add_library(${PROJECT_NAME} SHARED ${APP_SOURCES})
elseif(IOS)
	list(REMOVE_AT APP_SOURCES 0)
	list(APPEND APP_SOURCES
			${CMAKE_CURRENT_SOURCE_DIR}/ios/main.mm
			${CMAKE_CURRENT_SOURCE_DIR}/ios/AppDelegate.h
			${CMAKE_CURRENT_SOURCE_DIR}/ios/AppDelegate.m
			${CMAKE_CURRENT_SOURCE_DIR}/ios/ViewController.h
			${CMAKE_CURRENT_SOURCE_DIR}/ios/ViewController.mm
	)
    add_executable(${PROJECT_NAME} MACOSX_BUNDLE ${APP_SOURCES})
else()
    add_executable(${PROJECT_NAME} WIN32 ${APP_SOURCES})
endif()

target_link_libraries(${PROJECT_NAME} 
    PUBLIC 
    edge_framework
    application
)

set(PROJECT_OUTPUT_DIR "${CMAKE_BINARY_DIR}/bin/engine")
set(EMBED_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../embed")

add_custom_command(
    TARGET ${PROJECT_NAME} POST_BUILD 
    COMMAND ${CMAKE_COMMAND} -E make_directory "${PROJECT_OUTPUT_DIR}/embed"
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${EMBED_SOURCE_DIR}/font" "${PROJECT_OUTPUT_DIR}/embed/font"
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${EMBED_SOURCE_DIR}/mono" "${PROJECT_OUTPUT_DIR}/embed/mono"
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different "${EMBED_SOURCE_DIR}/textures" "${PROJECT_OUTPUT_DIR}/embed/textures"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EMBED_SOURCE_DIR}/shaders/technique_db.json" "${PROJECT_OUTPUT_DIR}/embed/technique_db.json"
)

if(MSVC)
    set_target_properties(engine PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${PROJECT_OUTPUT_DIR}
    )
    add_custom_command(TARGET engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_RUNTIME_DLLS:engine>
            $<TARGET_FILE_DIR:engine>
        COMMAND_EXPAND_LISTS
    )
endif()
