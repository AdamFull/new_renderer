cmake_minimum_required(VERSION 3.28.0)
project(engine)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(MAIN_PROJECT_DIRECTORY_PATH ${CMAKE_CURRENT_SOURCE_DIR})
set(OUT_BINARIES_PATH ${CMAKE_CURRENT_SOURCE_DIR}/bin)
set(ENGINE_BINARIES_PATH ${OUT_BINARIES_PATH}/${CMAKE_BUILD_TYPE}/engine)
set(TOOLS_BINARIES_PATH ${OUT_BINARIES_PATH}/${CMAKE_BUILD_TYPE}/tools)

file(MAKE_DIRECTORY ${OUT_BINARIES_PATH})

if(WIN32)
add_compile_definitions("PLATFORM_WIN32")
endif()

add_compile_definitions(FMT_UNICODE=0)

if(MSVC)
    add_compile_options(/bigobj)
endif()

add_compile_definitions(
    IMGUI_DEFINE_MATH_OPERATORS
    IMGUI_USE_WCHAR32
    NOMINMAX
    GLM_FORCE_INLINE 
    GLM_FORCE_XYZW_ONLY
    GLM_FORCE_CXX17
    GLM_FORCE_DEFAULT_ALIGNED_GENTYPES
    GLM_ENABLE_EXPERIMENTAL
    GLM_EXT_INCLUDED
)

# find_package(Taskflow CONFIG REQUIRED)
# target_link_libraries(main PRIVATE Taskflow::Taskflow)

# global includes
include_directories(
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_BINARY_DIR}/exports"
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/utllib/other"
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/utllib/source"
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/imdbg"
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/imgui"
    "${CMAKE_CURRENT_SOURCE_DIR}/libraries/edge_framework"
    "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/include"
)

add_compile_definitions($<$<CONFIG:DEBUG>:ENGINE_DEBUG>)

add_subdirectory(libraries/imdbg)
add_subdirectory(libraries/edge_framework/edge)

add_subdirectory(modules/glm)
add_subdirectory(modules/imgui)
add_subdirectory(libraries/imgui)
add_subdirectory(libraries/utllib)
add_subdirectory(tools/scene_compiler)

add_subdirectory(engine)

set(APP_SOURCES main.cpp)

if(ANDROID)
    if(CMAKE_VS_NsightTegra_VERSION)
        list(APPEND APP_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/android/AndroidManifest.xml)
    endif()

    add_library(${PROJECT_NAME} SHARED ${APP_SOURCES})
elseif(IOS)
	list(REMOVE_AT APP_SOURCES 0)
	list(APPEND APP_SOURCES
			${CMAKE_CURRENT_SOURCE_DIR}/ios/main.mm
			${CMAKE_CURRENT_SOURCE_DIR}/ios/AppDelegate.h
			${CMAKE_CURRENT_SOURCE_DIR}/ios/AppDelegate.m
			${CMAKE_CURRENT_SOURCE_DIR}/ios/ViewController.h
			${CMAKE_CURRENT_SOURCE_DIR}/ios/ViewController.mm
	)
    add_executable(${PROJECT_NAME} MACOSX_BUNDLE ${APP_SOURCES})
else()
    add_executable(${PROJECT_NAME} WIN32 ${APP_SOURCES})
endif()

target_link_libraries(${PROJECT_NAME} 
    PUBLIC imdbg
    PUBLIC imgui
    PUBLIC utility_lib
	PUBLIC application
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${ENGINE_BINARIES_PATH}
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${ENGINE_BINARIES_PATH}
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${ENGINE_BINARIES_PATH}
    RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL ${ENGINE_BINARIES_PATH}
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO ${ENGINE_BINARIES_PATH}
)
